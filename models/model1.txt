  
  model {
  
  # Priors
  # Uninformative
  psi ~ dunif(0,1)
  theta11 ~ dunif(0,1)
  p111 ~ dunif(0.5,1)
  p101 ~ dunif(0,0.5)
  p001 ~ dunif(0,0.5)
    
  
  # Likelihood
  for (i in 1:nsites) {
    # Ecological model for true occurrence
    z[i] ~ dbern(psi)
    
    for (j in 1:nindiv) {
      # Ecological model for true prevalence
      p1[i,j] ~ dbern(z[i]*theta11) # if z = 1
      
      w[i,j] <- ifelse(z[i] == 0, 0, p1[i,j])
      
      # Observation model
      # Sample detection state from categorical distribution
	    y[i,j] ~ dbinom(pi[i,j], ntests[i,j])
      
      # Define probability of detection
      pi[i,j] <- z[i]*w[i,j]*p111 + z[i]*(1-w[i,j])*p101 + (1-z[i])*(1-w[i,j])*p001
      
    } #j 
  } #i

}
