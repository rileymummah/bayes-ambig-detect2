
  model {
  
  # Priors
  # Uninformative
  psi ~ dunif(0,1)
  theta11 ~ dunif(0,1)
  b1 ~ dunif(delta,1)
  b2 ~ dunif(0, 1)
  b3 ~ dunif(0, 1)

  # Derived
  p111 <- delta/b1
  p101 <- 1-r
  p001 <- 1-r
  
  # Likelihood
  for (i in 1:nsites) {
    # Ecological model for true occurrence
    z[i] ~ dbern(psi)
          
    for (j in 1:nindiv) {
      # Ecological model for true prevalence
      p1[i,j] ~ dbern(z[i]*theta11) # if z = 1
      
      w[i,j] <- ifelse(z[i] == 0, 0, p1[i,j])
      
      ## Observation model
      # Define probabilities
      # P(Y = 0)
      pi[i,j,1] <- z[i]*w[i,j]*(1-p111) +
                   z[i]*(1-w[i,j])*(1-p101) + 
                   (1-z[i])*(1-w[i,j])*(1-p001)
      
      # P(Y = 1)            
      pi[i,j,2] <- z[i]*w[i,j]*b1*p111 + 
                   z[i]*(1-w[i,j])*b2*p101 +
                   (1-z[i])*(1-w[i,j])*b3*p001
      
      # P(Y = U)
      pi[i,j,3] <- z[i]*w[i,j]*(1-b1)*p111 + 
                   z[i]*(1-w[i,j])*(1-b2)*p101 + 
                   (1-z[i])*(1-w[i,j])*(1-b3)*p001
                   
      # Sample detection state from categorical distribution
      y[i,j,1:3] ~ dmulti(pi[i,j,1:3], ntests) #y=[0,1,U]
      
    } #j 
  } #i
  
} # model
