
  model {
  
  # Priors
  # Uninformative
  psi ~ dunif(0,1)
  theta11 ~ dunif(0,1)
  p111 ~ dunif(0.5,1)
  p100 ~ dunif(0.5,1)
  p000 ~ dunif(0.5,1)
  
  # Likelihood
  for (i in 1:nsites) {
    # Ecological model for true occurrence
    z[i] ~ dbern(psi) 
    p.eff[i] <- z[i] * theta11 + (1-z[i]) * theta01
    
    for (j in 1:nindiv) {
      # Ecological model for true prevalence
      w[i,j] ~ dbern(p.eff[i])
      
      # Observation model
      # Sample detection state from categorical distribution
      y[i,j,] ~ dmulti(pi[i,j,1:3], ntests) #y=[0,1,U]
      
      # Define probabilities
      pi[i,j,1] <- z[i]*w[i,j]*(p111*(1-delta) + (1-p111)*r) + 
                  z[i]*(1-w[i,j])*((1-p100)*(1-delta) + p100*r) + 
                  (1-z[i])*w[i,j]*(p011*(1-delta) + (1-p011)*r) + 
                  (1-z[i])*(1-w[i,j])*((1-p000)*(1-delta) + p000*r)
                  
      pi[i,j,2] <- z[i]*w[i,j]*(p111*delta + (1-p111)*(1-r)) + 
                  z[i]*(1-w[i,j])*((1-p100)*delta + p100*(1-r)) + 
                  (1-z[i])*w[i,j]*(p011*delta + (1-p011)*(1-r)) + 
                  (1-z[i])*(1-w[i,j])*((1-p000)*delta + p000*(1-r))
      
      pi[i,j,3] <- z[i]*w[i,j]*(p111*(1-delta) + (1-p111)*(1-r)) + 
                  z[i]*(1-w[i,j])*((1-p100)*(1-delta) + p100*(1-r)) + 
                  (1-z[i])*w[i,j]*(p011*(1-delta) + (1-p011)*(1-r)) + 
                  (1-z[i])*(1-w[i,j])*((1-p000)*(1-delta) + p000*(1-r))
      
    } #j 
  } #i
  
  # Derived quantities
  #occ.fs <- sum(z[])
  }
