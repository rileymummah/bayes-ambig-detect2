
  model {
  
  # Priors
  # Uninformative
  psi ~ dunif(0,1)
  theta11 ~ dunif(0,1)
  
  
  # Likelihood
  for (i in 1:nsites) {
    # Ecological model for true occurrence
    z[i] ~ dbern(psi) 
    p.eff[i] <- z[i] * theta11 + (1-z[i]) * theta01
    
    for (j in 1:nindiv) {
      # Ecological model for true prevalence
      w[i,j] ~ dbern(p.eff[i])
      
      # Observation model
      # Sample detection state from categorical distribution
	  y[i,j] ~ dbinom(pi[i,j], ntests[i,j])
      
      # Define probabilities [0,1,U]
      pi[i,j] <- w[i,j]*delta + (1-w[i,j])*(1-r)
      
    } #j 
  } #i

  }
